    precondition(AppleDock.apps(in: layout) == ["/Applications/Safari.app"], "dock layout")

    precondition(try! JSONDecoder().decode([DockThing].self, from: JSONEncoder().encode(things)) == things && things[3].title == "example.com", "dock items")
